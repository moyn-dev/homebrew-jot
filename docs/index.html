Static floor. Service ceiling.

Start with one file. Add code when the work needs rules.

A Service Jot can own request handling, durable state, logs, schedules, and policy without turning into a deployment project.

​​​
// server.js
export default function service(jot) {
  jot.fetch(async (request, env) => {
    const visits = (await env.state.get("visits") ?? 0) + 1;
    await env.state.put("visits", visits);
    return Response.json({ visits });
  });
}
